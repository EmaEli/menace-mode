@echo off
setlocal EnableExtensions EnableDelayedExpansion

set "ROOT=%~dp0"
set "PROJECT=%ROOT%src\LargerMaps\LargerMaps.csproj"
set "MODSRC=%ROOT%modpack\LargerMaps"
set "ARTIFACTS=%ROOT%artifacts"
set "STAGE=%ARTIFACTS%\nexus-stage"
set "ZIP=%ARTIFACTS%\LargerMaps-1.4.zip"
set "LOG=%ARTIFACTS%\build-nexus-release.log"

if not exist "%ARTIFACTS%" mkdir "%ARTIFACTS%"

if "%MENACE_GAME%"=="" set "MENACE_GAME=M:\SteamLibrary\steamapps\common\Menace"
set "GAME=%MENACE_GAME%"

> "%LOG%" echo Larger Maps 1.4 - Nexus release build
>>"%LOG%" echo Started: %date% %time%
>>"%LOG%" echo Working directory: %ROOT%
>>"%LOG%" echo GAME=%GAME%
>>"%LOG%" echo.

echo ============================================
echo   Larger Maps 1.4 - Nexus Release
echo ============================================
echo.
echo This builds a Nexus-ready ZIP and does not deploy game files.
echo.

if not exist "%GAME%\Menace.exe" (
  echo ERROR: MENACE not found at "%GAME%"
  echo Set MENACE_GAME to your game folder and run again.
  >>"%LOG%" echo ERROR: game folder not found.
  pause
  exit /b 2
)

if not exist "%GAME%\Mods\Menace.ModpackLoader.dll" (
  echo ERROR: Menace.ModpackLoader.dll not found in "%GAME%\Mods".
  >>"%LOG%" echo ERROR: Menace.ModpackLoader.dll missing.
  pause
  exit /b 3
)

where dotnet >nul 2>nul
if errorlevel 1 (
  echo ERROR: dotnet SDK not found.
  >>"%LOG%" echo ERROR: dotnet SDK not found.
  pause
  exit /b 4
)

rem Resolve Il2CppInterop across common MelonLoader layouts.
set "INTEROP_RUNTIME=%GAME%\MelonLoader\net6\Il2CppInterop.Runtime.dll"
if not exist "%INTEROP_RUNTIME%" set "INTEROP_RUNTIME=%GAME%\MelonLoader\Il2CppInterop.Runtime.dll"
if not exist "%INTEROP_RUNTIME%" (
  for /r "%GAME%\MelonLoader" %%F in (Il2CppInterop.Runtime.dll) do if not defined INTEROP_RUNTIME_FOUND set "INTEROP_RUNTIME_FOUND=%%~fF"
  if defined INTEROP_RUNTIME_FOUND set "INTEROP_RUNTIME=%INTEROP_RUNTIME_FOUND%"
)

set "INTEROP_COMMON=%GAME%\MelonLoader\net6\Il2CppInterop.Common.dll"
if not exist "%INTEROP_COMMON%" set "INTEROP_COMMON=%GAME%\MelonLoader\Il2CppInterop.Common.dll"
if not exist "%INTEROP_COMMON%" (
  for /r "%GAME%\MelonLoader" %%F in (Il2CppInterop.Common.dll) do if not defined INTEROP_COMMON_FOUND set "INTEROP_COMMON_FOUND=%%~fF"
  if defined INTEROP_COMMON_FOUND set "INTEROP_COMMON=%INTEROP_COMMON_FOUND%"
)

if not exist "%INTEROP_RUNTIME%" (
  echo ERROR: Il2CppInterop.Runtime.dll not found under "%GAME%\MelonLoader".
  >>"%LOG%" echo ERROR: Il2CppInterop.Runtime.dll not found.
  pause
  exit /b 4
)
if not exist "%INTEROP_COMMON%" (
  echo ERROR: Il2CppInterop.Common.dll not found under "%GAME%\MelonLoader".
  >>"%LOG%" echo ERROR: Il2CppInterop.Common.dll not found.
  pause
  exit /b 4
)

>>"%LOG%" echo INTEROP_RUNTIME=%INTEROP_RUNTIME%
>>"%LOG%" echo INTEROP_COMMON=%INTEROP_COMMON%
>>"%LOG%" echo.
>>"%LOG%" echo ==== dotnet --info ====
dotnet --info >>"%LOG%" 2>&1

>>"%LOG%" echo.
>>"%LOG%" echo ==== build ====
dotnet build "%PROJECT%" -c Release -p:GameDir="%GAME%" -p:InteropRuntime="%INTEROP_RUNTIME%" -p:InteropCommon="%INTEROP_COMMON%" >>"%LOG%" 2>&1
if errorlevel 1 (
  echo BUILD FAILED. See "%LOG%"
  type "%LOG%"
  pause
  exit /b 5
)

set "DLL=%ROOT%src\LargerMaps\bin\Release\net6.0\LargerMaps.dll"
if not exist "%DLL%" (
  echo ERROR: Release DLL was not produced at "%DLL%".
  >>"%LOG%" echo ERROR: release DLL missing after build.
  pause
  exit /b 6
)

if exist "%STAGE%" rmdir /s /q "%STAGE%"
if exist "%ZIP%" del /q "%ZIP%"
mkdir "%STAGE%\LargerMaps\dlls"

copy /Y "%MODSRC%\modpack.json" "%STAGE%\LargerMaps\modpack.json" >>"%LOG%" 2>&1
if errorlevel 1 goto :package_failed
copy /Y "%DLL%" "%STAGE%\LargerMaps\dlls\LargerMaps.dll" >>"%LOG%" 2>&1
if errorlevel 1 goto :package_failed

powershell -NoProfile -ExecutionPolicy Bypass -Command "Compress-Archive -Path '%STAGE%\LargerMaps' -DestinationPath '%ZIP%' -Force" >>"%LOG%" 2>&1
if errorlevel 1 goto :package_failed

if not exist "%ZIP%" goto :package_failed

for %%F in ("%ZIP%") do set "ZIP_SIZE=%%~zF"
>>"%LOG%" echo.
>>"%LOG%" echo SUCCESS: %ZIP%
>>"%LOG%" echo ZIP bytes: !ZIP_SIZE!

echo.
echo SUCCESS: Nexus package created:
echo   "%ZIP%"
echo.
echo Contents:
echo   LargerMaps\modpack.json
echo   LargerMaps\dlls\LargerMaps.dll
echo.
echo Nothing was deployed to MENACE.
pause
exit /b 0

:package_failed
echo PACKAGE FAILED. See "%LOG%"
>>"%LOG%" echo ERROR: package creation failed.
type "%LOG%"
pause
exit /b 7
